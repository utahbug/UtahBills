$(document).ready(function() {
		$('.trackchkbox').each(function () {
			//console.log($(this).attr('id'));
			//trkItems[]
			var cookieValue = $.cookie("BillList");
			if(cookieValue!=null){
				if(cookieValue.indexOf($(this).attr("data-trackingid"))>=0){
					//console.log($(this).val());
					$(this).prop('checked', true);
				}
			}
		});
		loadBillListCount();
	});

	function trackClick(billnum,thischk){
		console.log($(thischk).is(':checked'));
		var trackingId = $(thischk).data('trackingid');
		if ($(thischk).is(':checked')) {
			trackBill(billnum, trackingId);
		} else {
			stopTrackingBill(billnum, trackingId);
		}
	}

	function loadBillListCount(){
		var str = "Bills being tracked: ";
		var tcnt = 0;
		var billList = localStorage.getItem("trackBillsList");
		if(billList!=null){
			var trackarr = billList.split(",");
			// $("input[type=checkbox]").prop('checked', false);
			$("input.trackchkbox").prop('checked', false);
			for (i=0;i<trackarr.length;i++){
				//$("#trk"+trackarr[i]).prop('checked', true);
				$("input[id=trk"+trackarr[i]+"]").each(function(index,elem){
					$(elem).prop('checked', true);
				});

				if(trackarr[i]!=""){
					tcnt++;
				}
			}
		}
		str+=tcnt;
		$('.tracklink').html(str);
	}

	function trackBill(blist, trackingId){
		$.ajax({
			method: "POST",
			url: "/tracking/trackingAddBills.jsp",
			data: { bills: blist }
		})
			.done(function( msg ) {
			if(msg.trim()=="success"){
				var billList = localStorage.getItem("trackBillsList");
				if(billList==null){ billList=""; };
				if(billList!=""){billList+=",";}
				billList+=trackingId;
				localStorage.setItem("trackBillsList",billList);
				loadBillListCount();
			}else{
			}
		});
	}

	function stopTrackingBill(blist, trackingId){
		$.ajax({
			method: "POST",
			url: "/tracking/trackingRemoveBill.jsp",
			data: { bill: blist }
		})
			.done(function( msg ) {
			if(msg.trim()=="success"){
				var billList = localStorage.getItem("trackBillsList");
				if(billList==null){ billList=""; }
				// billList=billList.replace(","+blist,"");
				// billList=billList.replace(blist,"");
				var trackarr = billList.split(",");
                var newarr = [];
                for (i=0;i<trackarr.length;i++){
                    if(trackarr[i]!="" && trackarr[i]!=trackingId){
                        newarr.push(trackarr[i]);
					}
				}

				localStorage.setItem("trackBillsList",newarr.join(","));
                loadBillListCount();

				// localStorage.setItem("trackBillsList",billList);
				// loadBillListCount();
			}else{
			}
		});
	}
